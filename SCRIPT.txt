end

	if isDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		local delta = input.Position - dragStartPosition
		MainFrame.Position = UDim2.new(
			frameStartPosition.X.Scale, 
			frameStartPosition.X.Offset + delta.X, 
			frameStartPosition.Y.Scale,

		)
	end
end)

		isDragging = false
	end

local function addToggleSetting(label, getValue, onToggle)
	local box = createSettingBox(label, 40)

	local toggleBtn = Instance.new("TextButton", box)
	toggleBtn.Size = UDim2.new(0, 42, 0, 20)
	toggleBtn.Position = UDim2.new(1, -52, 0.5, -10)
	toggleBtn.BackgroundColor3 = getValue() and Color3.fromRGB(0, 220, 150) or Color3.fromRGB(32, 38, 50)
	toggleBtn.Text = ""

	local btnCorner = Instance.new("UICorner", toggleBtn)
	btnCorner.CornerRadius = UDim.new(1, 0)

	local indicator = Instance.new("Frame", toggleBtn)
	indicator.Size = UDim2.new(0, 14, 0, 14)
	indicator.Position = getValue() and UDim2.new(1, -17, 0.5, -7) or UDim2.new(0, 3, 0.5, -7)
	indicator.BackgroundColor3 = Color3.fromRGB(255, 255, 255)

	local indCorner = Instance.new("UICorner", indicator)
	indCorner.CornerRadius = UDim.new(1, 0)

	local function refreshVisuals()
		local active = getValue()
		TweenService:Create(toggleBtn, TweenInfo.new(0.2), {
			BackgroundColor3 = active and Color3.fromRGB(0, 220, 150) or Color3.fromRGB(32, 38, 50)
		}):Play()
		TweenService:Create(indicator, TweenInfo.new(0.2), {
			Position = active and UDim2.new(1, -17, 0.5, -7) or UDim2.new(0, 3, 0.5, -7)
		}):Play()
	end

	toggleBtn.MouseButton1Click:Connect(function() 
		onToggle() 
		refreshVisuals() 
	end)

	RegisteredToggles[getValue] = refreshVisuals
end

		ColorSequenceKeypoint.new(0.17, Color3.fromRGB(255, 255, 0)),

				applyNewColor()
			end
		end
	end)

local function clearPlayerVisuals(character)
	if State.TrackedPlayers[character] then
		for _, visualObject in pairs(State.TrackedPlayers[character]) do
			if typeof(visualObject) == "Instance" then 
				visualObject:Destroy() 
			elseif typeof(visualObject) == "table" then
				for _, subObject in pairs(visualObject) do
					if typeof(subObject) == "Instance" then subObject:Destroy() end
				end
			end
		end
		State.TrackedPlayers[character] = nil
	end
end

function createHighlight(character, targetPlayer)
	clearPlayerVisuals(character)

	if Config.TeamCheck and targetPlayer.Team and LocalPlayer.Team and targetPlayer.Team == LocalPlayer.Team then
		return
	end

	local playerTrackData = {}

	if Config.ESPType == "Highlight" then
		local highlight = Instance.new("Highlight")
		highlight.Name = generateInterfaceName()
		highlight.Adornee = character
		highlight.OutlineTransparency = (Config.HighlightStyle == "Fill") and 1 or 0
		highlight.FillTransparency = (Config.HighlightStyle == "Outline") and 1 or Config.Transparency
		highlight.OutlineColor = Config.OutlineColor
		highlight.FillColor = Config.HighlightColor
		highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		highlight.Parent = character
		playerTrackData.Highlight = highlight

	elseif Config.ESPType == "SelectionBox" then
		playerTrackData.Boxes = {}
		for _, part in pairs(character:GetChildren()) do
			if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
				local boxAdornment = Instance.new("BoxHandleAdornment")
				boxAdornment.Name = generateInterfaceName()
				boxAdornment.Adornee = part
				boxAdornment.Size = part.Size + Vector3.new(0.05, 0.05, 0.05)
				boxAdornment.Color3 = Config.OutlineColor
				boxAdornment.Transparency = (Config.HighlightStyle == "Outline") and 1 or Config.Transparency
				boxAdornment.AlwaysOnTop = true
				boxAdornment.ZIndex = 5
				boxAdornment.Parent = part

				table.insert(playerTrackData.Boxes, boxAdornment)
			end
		end
	end

	if Config.ShowPlayerTrails and character:FindFirstChild("HumanoidRootPart") then
		local rootPart = character.HumanoidRootPart

		local attachment0 = Instance.new("Attachment", rootPart)
		attachment0.Position = Vector3.new(-0.4, -2.8, 0)

		local attachment1 = Instance.new("Attachment", rootPart)
		attachment1.Position = Vector3.new(0.4, -2.8, 0)

		local trail = Instance.new("Trail")
		trail.Name = generateInterfaceName()
		trail.Attachment0 = attachment0
		trail.Attachment1 = attachment1
		trail.Lifetime = 1.2
		trail.MinLength = 0.1
		trail.Color = ColorSequence.new(Config.TrailColor)
		trail.Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0.2),
			NumberSequenceKeypoint.new(1, 1)
		})
		trail.Parent = rootPart

		playerTrackData.Trail = trail
		playerTrackData.Attachment0 = attachment0
		playerTrackData.Attachment1 = attachment1
	end

	if Config.ShowInfo and character:FindFirstChild("Head") then
		local billboard = Instance.new("BillboardGui")
		billboard.Name = generateInterfaceName()
		billboard.Adornee = character.Head
		billboard.Size = UDim2.new(0, 160, 0, 50)
		billboard.StudsOffset = Vector3.new(0, 2.8, 0)
		billboard.AlwaysOnTop = true
		billboard.MaxDistance = math.huge

		local function createCleanTextLabel(parent, size, pos, fontSize, defaultColor)
			local label = Instance.new("TextLabel", parent)
			label.Size = size
			label.Position = pos
			label.BackgroundTransparency = 1
			label.Font = Enum.Font.SourceSansBold
			label.TextSize = fontSize
			label.TextColor3 = defaultColor
			label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
			label.TextStrokeTransparency = 0
			return label
		end

		local nameLabel = createCleanTextLabel(billboard, UDim2.new(1, 0, 0, 15), UDim2.new(0, 0, 0, 0), 14, Color3.fromRGB(255, 255, 255))
		nameLabel.Text = targetPlayer.DisplayName

		local distanceLabel = createCleanTextLabel(billboard, UDim2.new(1, 0, 0, 14), UDim2.new(0, 0, 0, 16), 13, Color3.fromRGB(0, 220, 255))
		distanceLabel.Name = "DistanceLabel"
		distanceLabel.Text = "0 studs"

		local healthLabel = createCleanTextLabel(billboard, UDim2.new(1, 0, 0, 14), UDim2.new(0, 0, 0, 31), 13, Color3.fromRGB(0, 255, 120))
		healthLabel.Name = "HealthLabel"
		healthLabel.Text = "100 HP"

		billboard.Parent = character
		playerTrackData.Billboard = billboard
	end

	if Config.ShowTracers and character:FindFirstChild("HumanoidRootPart") then
		local tracerLine = Instance.new("LineHandleAdornment")
		tracerLine.Name = generateInterfaceName()
		tracerLine.Length = 0
		tracerLine.Thickness = 2
		tracerLine.Color3 = Config.HighlightColor
		tracerLine.AlwaysOnTop = true
		tracerLine.Adornee = character.HumanoidRootPart
		tracerLine.Parent = character

		playerTrackData.Tracer = tracerLine
	end

	State.TrackedPlayers[character] = playerTrackData
end

function enableESP()
	disableESP()
	if not Config.ESPEnabled then return end
	for _, player in pairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character then
			createHighlight(player.Character, player)
		end
	end
end

function disableESP()
	for character, _ in pairs(State.TrackedPlayers) do
		clearPlayerVisuals(character)
	end
	State.TrackedPlayers = {}
end

function updateAllHighlights()
	if not Config.ESPEnabled then return end
	for character, trackData in pairs(State.TrackedPlayers) do
		if trackData.Highlight then
			trackData.Highlight.OutlineTransparency = (Config.HighlightStyle == "Fill") and 1 or 0
			trackData.Highlight.FillTransparency = (Config.HighlightStyle == "Outline") and 1 or Config.Transparency
			trackData.Highlight.OutlineColor = Config.OutlineColor
			trackData.Highlight.FillColor = Config.HighlightColor
		end
		if trackData.Boxes then
			for _, box in pairs(trackData.Boxes) do
				box.Color3 = Config.OutlineColor
				box.Transparency = (Config.HighlightStyle == "Outline") and 1 or Config.Transparency
			end
		end
		if trackData.Trail then
			trackData.Trail.Color = ColorSequence.new(Config.TrailColor)
		end
		if trackData.Tracer then
			trackData.Tracer.Color3 = Config.HighlightColor
		end
	end
end

local function setupPlayerListeners(player)
	if player ~= LocalPlayer then
		player.CharacterAdded:Connect(function(char)
			if Config.ESPEnabled then
				task.wait(0.5)
				createHighlight(char, player)
			end
		end)
		if player.Character and Config.ESPEnabled then 
			createHighlight(player.Character, player) 
		end
	end
end

for _, player in pairs(Players:GetPlayers()) do setupPlayerListeners(player) end
Players.PlayerAdded:Connect(setupPlayerListeners)
Players.PlayerRemoving:Connect(function(player)
	if player.Character then clearPlayerVisuals(player.Character) end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if State.ActiveKeyBind and input.UserInputType == Enum.UserInputType.Keyboard then
		State.ActiveKeyBind.set(input.KeyCode)
		State.ActiveKeyBind.button.Text = input.KeyCode.Name
		State.ActiveKeyBind = nil
		return
	end
	if input.KeyCode == Config.ToggleKey then
		Config.ESPEnabled = not Config.ESPEnabled
		if Config.ESPEnabled then enableESP() else disableESP() end
		for _, refreshFunc in pairs(RegisteredToggles) do refreshFunc() end
	elseif input.KeyCode == Config.GUIToggleKey then
		Config.GUIVisible = not Config.GUIVisible
		MainFrame.Visible = Config.GUIVisible
	end
end)

RunService.RenderStepped:Connect(function()
	State.HuePosition = (State.HuePosition + 0.003) % 1
	local currentRainbowColor = Color3.fromHSV(State.HuePosition, 0.9, 1)

	local needsUpdate = false

	if Config.RainbowHighlight then
		Config.HighlightColor = currentRainbowColor
		needsUpdate = true
	end
	if Config.RainbowOutline then
		Config.OutlineColor = currentRainbowColor
		needsUpdate = true
	end
	if Config.RainbowTrail then
		Config.TrailColor = currentRainbowColor
		needsUpdate = true
	end

	if needsUpdate then
		updateAllHighlights()
	end

	if not Config.ESPEnabled then return end

	local myCharacter = LocalPlayer.Character
	local myRoot = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")

	for character, trackData in pairs(State.TrackedPlayers) do
		if character and character.Parent then
			local humanoid = character:FindFirstChildOfClass("Humanoid")
			local rootPart = character:FindFirstChild("HumanoidRootPart")

			if trackData.Billboard and humanoid and rootPart and myRoot then
				local distLabel = trackData.Billboard:FindFirstChild("DistanceLabel")
				if distLabel then
					if Config.ShowDistance then
						local distance = math.floor((myRoot.Position - rootPart.Position).Magnitude)
						distLabel.Text = tostring(distance) .. " studs"
						distLabel.Visible = true
					else
						distLabel.Visible = false
					end
				end

				local hpLabel = trackData.Billboard:FindFirstChild("HealthLabel")
				if hpLabel then
					if Config.ShowHealth then
						hpLabel.Visible = true
						local currentHp = math.floor(humanoid.Health)
						local percentHp = math.floor((humanoid.Health / humanoid.MaxHealth) * 100)
						hpLabel.Text = string.format("%d HP / %d%%", currentHp, percentHp)

						if percentHp > 60 then
							hpLabel.TextColor3 = Color3.fromRGB(0, 255, 120)
						elseif percentHp > 30 then
							hpLabel.TextColor3 = Color3.fromRGB(255, 200, 0)
						else
							hpLabel.TextColor3 = Color3.fromRGB(255, 60, 60)
						end
					else
						hpLabel.Visible = false
					end
				end
			end
		else
			clearPlayerVisuals(character)
		end
	end
end)
